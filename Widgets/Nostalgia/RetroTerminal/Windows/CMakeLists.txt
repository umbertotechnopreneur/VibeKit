cmake_minimum_required(VERSION 3.25)
project(RetroTerminal LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Build with MSVC and the Windows 11 SDK on Windows.")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(recovery-native-worker ../../../Shared/tests/recovery-native-worker.cpp)
    target_compile_features(recovery-native-worker PRIVATE cxx_std_20)
    target_compile_definitions(recovery-native-worker PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(recovery-native-worker PRIVATE /W4 /EHsc /permissive- /utf-8)
    foreach(check IN ITEMS settings persistence registry presentation)
        add_executable(${check}-checks tests/${check}-checks.cpp)
        target_compile_features(${check}-checks PRIVATE cxx_std_20)
        target_compile_definitions(${check}-checks PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
        # Assertions must remain active in the Release configuration used by CI.
        target_compile_options(${check}-checks PRIVATE /W4 /WX /EHsc /permissive- /utf-8 /UNDEBUG)
        add_test(NAME ${check}-checks COMMAND ${check}-checks)
    endforeach()
    target_link_libraries(registry-checks PRIVATE advapi32)
endif()

add_library(RetroEngine SHARED RetroEngine.cpp Renderer.cpp)
set_property(TARGET RetroEngine PROPERTY CXX_STANDARD 20)
set_property(TARGET RetroEngine PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(RetroEngine PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(RetroEngine PRIVATE /W4 /EHsc /permissive- /utf-8)
target_link_libraries(RetroEngine PRIVATE d3d11 dxgi dcomp d3dcompiler windowsapp user32 wtsapi32 advapi32 comctl32 shell32)
