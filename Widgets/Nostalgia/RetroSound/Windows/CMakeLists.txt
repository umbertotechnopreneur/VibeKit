cmake_minimum_required(VERSION 3.25)
project(RetroSound LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Build with MSVC and the Windows SDK.")
endif()

add_library(RetroSound SHARED Main.cpp ../../Shared/Native/Engine.cpp ../Shared/Mixer.cpp)
target_include_directories(RetroSound PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(RetroSound PRIVATE cxx_std_20)
target_compile_definitions(RetroSound PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(RetroSound PRIVATE /W4 /EHsc /utf-8)
set_property(TARGET RetroSound PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(RetroSound PRIVATE user32 gdi32 shell32 ole32 advapi32 winmm pdh comctl32 windowsapp wtsapi32)
set_target_properties(RetroSound PROPERTIES OUTPUT_NAME "WidgetEngine")
install(TARGETS RetroSound RUNTIME DESTINATION .)
install(DIRECTORY ../Shared/Samples/classic/ DESTINATION default FILES_MATCHING PATTERN "*.wav")
install(DIRECTORY ../Shared/Samples/warm/ DESTINATION warm FILES_MATCHING PATTERN "*.wav")
install(DIRECTORY ../Shared/Samples/computer/ DESTINATION computer FILES_MATCHING PATTERN "*.wav")
install(FILES ../LICENSE DESTINATION .)
